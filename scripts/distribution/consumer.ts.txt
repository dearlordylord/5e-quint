import assert from "node:assert/strict";
import * as creation from "@dearlordylord/dnd-sdk/character-creation";
import * as sheet from "@dearlordylord/dnd-sdk/character-sheet";
import * as characterBattle from "@dearlordylord/dnd-sdk/character-battle";
import * as battle from "@dearlordylord/dnd-sdk/battle";
import { srdSurface } from "@dearlordylord/dnd-sdk/catalog";

const draft = creation.createCharacterDraft({});
assert.deepEqual(draft.selections.choices, []);
assert.equal(draft.revision, 0);
for (const api of [creation, sheet, characterBattle, battle]) {
  assert(Object.keys(api).length > 0);
}
assert(srdSurface.units.length > 0);
assert(srdSurface.statBlocks.length > 0);
